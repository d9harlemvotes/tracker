The ID is the long string in the sheet URL between // /d/ and /edit: docs.google.com/spreadsheets/d/THIS_IS_THE_ID/edit var ENGAGEMENT_SHEET_ID = 'PASTE_NEW_SHEET_ID_HERE'; var ENGAGEMENT_TAB = 'Voter Engagement'; var ENGAGEMENT_HEADERS = [ 'Timestamp', 'Shift Date', 'Site', 'Volunteer Name', 'Organization', 'Path', 'Notes' ]; function doGet() { return ContentService.createTextOutput('D9 Harlem Votes voter engagement endpoint is live.'); } function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(20000); } catch (lockErr) {} try { var data = {}; if (e && e.postData && e.postData.contents) data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.openById(ENGAGEMENT_SHEET_ID); var tab = getTab_(ss, ENGAGEMENT_TAB, ENGAGEMENT_HEADERS); tab.appendRow([ new Date(), data.shiftDate || '', data.siteName || '', data.volunteerName || '', data.organization || '', data.path || '', data.notes || '' ]); return json_({ result: 'success', tab: ENGAGEMENT_TAB }); } catch (error) { try { getTab_(SpreadsheetApp.openById(ENGAGEMENT_SHEET_ID), 'Engagement Errors', ['Timestamp', 'Error', 'Raw JSON']) .appendRow([ new Date(), error.toString(), e && e.postData ? e.postData.contents : '' ]); } catch (ignore) {} return json_({ result: 'error', message: error.toString() }); } finally { try { lock.releaseLock(); } catch (relErr) {} } } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Get a tab by name, creating it with headers if missing.
